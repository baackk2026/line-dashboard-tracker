const LIFF_ID = "2011130854-3vn9oGZQ"; const APPS_SCRIPT_URL = "https://script.google.com/macros/s/AKfycbyLJiU0vuYhqXdQTAatMVtsMI_z5g1m-jytaHXJHufNCnQYPU3MtMsRwDVN7KnstyCM/exec"; const DASHBOARDS = { "kpi": { name: "KPI" }, "4dx": { name: "4DX" }, "performance": { name: "ผลประกอบการ" } }; function showLoading(msg) { document.getElementById("app").innerHTML = `

`; } function showError(title, msg) { document.getElementById("app").innerHTML = `
⚠️
${title}
${msg}
`; } function getDashboardId() { const p = new URLSearchParams(window.location.search); const direct = p.get("dashboard"); if (direct) return direct; const liffState = p.get("liff.state"); if (liffState) { try { const decoded = decodeURIComponent(liffState); const s = decoded.startsWith("?") ? decoded.slice(1) : decoded; return new URLSearchParams(s).get("dashboard"); } catch(e) {} } return sessionStorage.getItem("pending_dashboard"); } async function main() { try { showLoading("กำลังโหลด..."); const earlyId = getDashboardId(); if (earlyId) sessionStorage.setItem("pending_dashboard", earlyId); await liff.init({ liffId: LIFF_ID }); const dashboardId = getDashboardId(); if (!dashboardId || !DASHBOARDS[dashboardId]) { showError("ไม่พบข้อมูล Dashboard", "กรุณาเข้าผ่าน Rich Menu ของ LINE เท่านั้น"); return; } if (!liff.isLoggedIn()) { const base = window.location.origin + window.location.pathname; liff.login({ redirectUri: `${base}?dashboard=${dashboardId}` }); return; } showLoading(`กำลังเข้าสู่ ${DASHBOARDS[dashboardId].name}...`); const profile = await liff.getProfile(); const userId = profile.userId; sessionStorage.removeItem("pending_dashboard"); // Redirect ไป Apps Script — Apps Script log แล้ว redirect ต่อไป Looker window.location.href = `${APPS_SCRIPT_URL}?userId=${encodeURIComponent(userId)}&dashboardId=${encodeURIComponent(dashboardId)}`; } catch(err) { console.error(err); showError("เกิดข้อผิดพลาด", "กรุณาปิดแล้วเปิดใหม่อีกครั้ง"); } } main();
